import joblib
import pandas as pd
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import top_k_accuracy_score, ...

TOP_K = 4
FEATURES = ["Weekday", "Month_name", "Time_of_Day"]
TARGET = "coffee_name"

WEEKDAY_ORDER = ["Mon", "Tue", ..., "Sun"]
MONTH_ORDER   = ["Jan", "Feb", ..., "Dec"]

ตั้งค่าจุดเดียว ใช้ทั้งไฟล์

  • FEATURES คือ 3 คอลัมน์ที่โมเดล "รู้ล่วงหน้าได้จริง" ก่อนลูกค้าสั่งเมนู
  • ตัด money, cash_type, hour_of_day ทิ้งตั้งแต่ตรงนี้ — ไม่ถูกอ้างถึงอีกเลย
  • WEEKDAY_ORDER / MONTH_ORDER เก็บลำดับที่ถูกต้อง (จ.-อา.) ไว้ให้หน้าเว็บใช้สร้าง dropdown

build_pipeline() — แปลงข้อความเป็นตัวเลข

preprocessor = ColumnTransformer(
    transformers=[
        ("cat", OneHotEncoder(handle_unknown="ignore"), FEATURES)
    ]
)

Weekday="Mon" → [1,0,0,0,0,0,0]

  • OneHotEncoder แปลงแต่ละหมวดหมู่เป็นเวกเตอร์ 0/1 แยกช่อง
  • handle_unknown="ignore" คือประกันภัย — เจอค่าแปลกที่ไม่เคยเห็นตอนเทรน จะได้เวกเตอร์ 0 ล้วน แทนที่จะพัง
train_model.py บรรทัด 50-51

Pipeline — ห่อ 2 ขั้นตอนเป็นวัตถุเดียว

clf = GradientBoostingClassifier(random_state=42)

return Pipeline(steps=[
    ("preprocess", preprocessor),
    ("model", clf),
])
  • เรียก .fit() ครั้งเดียว — รัน encoder ก่อน แล้วค่อยเทรนโมเดล
  • เรียก .predict() ทีหลัง — แปลงข้อมูลใหม่แบบเดียวกับตอนเทรนโดยอัตโนมัติ
  • ป้องกันบั๊กยอดฮิต: ลืมแปลงข้อมูล inference ให้เหมือนตอนเทรน

ส่งกลับเป็น JSON พร้อมป้าย "ติดท็อป 4 ไหม"

return jsonify({
    "prediction": ranked[0][0],
    "top_k": top_k,
    "probabilities": [
        {"coffee_name": name, "probability": round(float(p), 4),
         "in_top_k": rank < top_k}
        for rank, (name, p) in enumerate(ranked)
    ],
})
  • rank < top_k → True/False บอกว่าอยู่ใน 4 อันดับแรกไหม

  • float(p) แปลงจาก numpy type เป็น float ปกติ ไม่งั้น jsonify พังตอนแปลง JSON
